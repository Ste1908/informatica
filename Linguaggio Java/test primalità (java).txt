package progetto_java;

import java.util.Scanner;

public class codice_prova {

	public static void main(String[] args) {
		
		Scanner scanner = new Scanner(System.in);
		System.out.println("inserisci un numero");
		int numero = scanner.nextInt();
		int z=0;
		int i=3;
		float limite= numero/2;
        do {
        	if(numero%i==0||numero%2==0||numero==0||numero==1) {
        		z=1; i=numero;
        	}i+=2;
        }while(i<limite);
        if(numero==2) z=0;
        if(z==1) System.out.println(numero+ " non è un numero primo");
        else System.out.println(numero+ " è un numero primo");
	}
	}
